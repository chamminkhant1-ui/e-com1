      const yearCheck = document.getElementById("yearMessage");
      const rollNumberCheck = document.getElementById("rollNumberMessage");

      const yearText = yearCheck ? yearCheck.textContent.trim() : "";
      const rollText = rollNumberCheck ? rollNumberCheck.textContent.trim() : "";
      const mailText = mailCheck ? mailCheck.textContent.trim() : "";

      // ✅ Condition: Only allow submit when all match
      const isNRCValid = nrcText === "✅";
      const isYearValid = yearText === "";
      const isRollValid = rollText === "";
      const isMailValid = mailText !== "Email domain/server not verified.";

      if (isNRCValid && isYearValid && isRollValid && isMailValid) {
        // ✅ All checks passed → proceed to /RegistrationConfirmation
        this.action = "/RegistrationConfirmation";
      } else {
        // ❌ Stop submission
        event.preventDefault();
        let messages = [];

        if (!isNRCValid) messages.push("NRC check failed ❌");
        if (!isYearValid) messages.push("Matriculation year check failed ❌");
        if (!isRollValid) messages.push("Roll number check failed ❌");
        if (!isMailValid) messages.push("Email verification failed ❌");

        alert("Please fix the following before submitting:\n\n" + messages.join("\n"));
      }
    });

    function toMyanmarNumber(engNumber) {
      const myanmarDigits = ["၀","၁","၂","၃","၄","၅","၆","၇","၈","၉"];
      return engNumber.toString().replace(/[0-9]/g, d => myanmarDigits[d]);

        const formattedDate = `${day}-${month}-${year}`;

    const values = [r1.value, r2.value, r3.value].filter(v => v);
    output.value = values.join(' / ');
  }

  function loadDistricts(stateId, districtId, townshipId, addressId) {
    const stateSelect = document.getElementById(stateId);

    const townshipSelect = document.getElementById(townshipId);
    const addressSelect = document.getElementById(addressId);

      return;
    }

      .then(data => {
        let optionHTML = '<option value="">--ရွေးချယ်ပါ--</option>';

        townshipSelect.value="";

        addressSelect.value="";
        addressSelect.disabled = true;
      })
      .catch(error => {
        console.error("Error loading districts:", error);
        districtSelect.innerHTML = '<option value="">---</option>';
      });
  }

  async function checkEmail() {
    const emailInput = document.getElementById("std_email");
    const message = document.getElementById("mailMessage");
    const email = emailInput.value.trim();


    // Send to backend for validation
    try {
      const response = await fetch("/check_email", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email })
      });

      const data = await response.json();

      if (!data.is_valid) {
        message.textContent = "Invalid email format.";
        message.style.color = "red";
      } else if (!data.is_verified) {
        message.textContent = "Email domain/server not verified.";
        message.style.color = "orange";
      } else {
        message.textContent = "Valid and verified email.";
        message.style.color = "green";
      }
    } catch (err) {
      console.error(err);
      message.textContent = "Error checking email.";
      message.style.color = "red";
    }
  }
